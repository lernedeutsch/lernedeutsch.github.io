function evaluate(answer){ const text = answer.toLowerCase(); /* Jak masz na imię? */ if(lessonStep === 1){ if( text.includes("ich heiße") || text.includes("mein name ist") || text.includes("ich bin") ){ let name = ""; const match = answer.match(/ich heiße\s+([a-zäöüß]+)/i) || answer.match(/mein name ist\s+([a-zäöüß]+)/i); if(match){ name = ", " + match[1]; } addBot( "Freut mich" + name + ". Wie geht es dir?" ); lessonStep = 2; }else{ addBot( "Ich habe deinen Namen nicht verstanden. Du kannst zum Beispiel sagen: Ich heiße Anna. Wie heißt du?" ); } return; } /* Jak się masz? */ if(lessonStep === 2){ if( text.includes("gut") || text.includes("sehr gut") || text.includes("prima") || text.includes("super") || text.includes("es geht") ){ addBot( "Mir geht es auch gut, danke. Ich muss jetzt leider los." ); lessonStep = 3; }else{ addBot( "Ich habe dich nicht ganz verstanden. Du kannst zum Beispiel sagen: Gut, danke. Wie geht es dir?" ); } return; } /* Naturalne pożegnanie */ if(lessonStep === 3){ if( text.includes("tschüss") || text.includes("auf wiedersehen") || text.includes("bis später") || text.includes("bis bald") ){ addBot( "Tschüss!
